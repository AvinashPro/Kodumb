1. createCanvas()
function createCanvas(w, h) {
  const dpr = window.devicePixelRatio || 1;
  const canvas = document.createElement("canvas");
  
  canvas.width = w * dpr;
  canvas.height = h * dpr;
  canvas.style.width = w + "px";
  canvas.style.height = h + "px";
  
  const ctx = canvas.getContext("2d");
  ctx.imageSmoothingEnabled = false;
  ctx.scale(dpr, dpr);
  
  return canvas;
}
This function helps us to create dpr scaled canvas.

Note

class AssetLoader {
  constructor() {
    this.assets = new Map();
  }
  
  load(data = {}, callback) {
    
    this.itemsLoaded = 0;
    this.itemsToLoad = Object.keys(data).length;
    
    if(this.itemsToLoad == 0) {
      callback();
      return;
    }
    
    for(const [name, props] of Object.entries(data)) {
      this[`load${props.type}`](props.url).then(value => {
        
        this.assets.set(name, value);
        this.itemsLoaded += 1;
        this.progress = this.itemsLoaded / this.itemsToLoad;
        
        if(this.progress === 1) {
          callback();
        }
        
      }).catch(() => {
        console.log("Some error occured!")
      })
      
    }
    
  }
  
  loadImg(url) {
    return new Promise((resolve, reject) => {
      let img = new Image();
      img.src = url;
      img.onload = () => {
        resolve(img);
      }
      img.onerror = () => {
        reject();
      }
    })
  }
  
  loadJSON(url) {
    return new Promise((resolve, reject) => {
      let request = fetch(url);
      request.then(response => {
        resolve(response.json());
      }).catch(() => {
        reject();
      })
    })
  }
  
  loadAudio(url) {
    return new Promise((resolve, reject) => {
      let audio = new Audio();
      audio.src = url;
      audio.oncanplay = () => {
        resolve(audio);
      }
      audio.onerror = () => {
        reject();
      }
    })
  }
  
  get(name) {
    return this.assets.get(name);
  }
  
}

3. Vector class
It's our last utility but a very important one! I'll give you a simple explanation (which is not quite accurate but serves our purpose for now). In physics, vectors are quantities that have both direction and magnitude. For example, consider a car going 5 km per hour (magnitude). But where? what's the direction? We see, velocity's description is pointless unless we define its direction, say north or 34° from +x-axis (direction) with respect to our chosen coordinate system.

"Why we need it?" you might ask. In a game, we have many entities such players, enemies, projectiles, tiles or background. Many entities have vector quantities such as position (x, y), velocity, acceleation etc. It becomes quite a nightmare if you don't find a clean way to organise and work with them. That's why we use vectors to represent them, store their data and perform logic (update)
class Vector {
  constructor(x = 0, y = 0) {
    this.x = x;
    this.y = y;
  }
  
  // to add another vector
  add(vec) {
    this.x += vec.x;
    this.y += vec.y;
    return this;
  }
  
  // to subtract another vector
  sub(vec) {
    this.x -= vec.x;
    this.y -= vec.y;
    return this;
  }
  
  // to scale vector by a given factor
  mult(k) {
    this.x *= k;
    this.y *= k;
    return this;
  }
  
  // to set our vector to given (x,y) components
  set(x, y) {
    this.x = x;
    this.y = y;
    return this;
  }
  
  // magnitude of vector
  mag() {
    return Math.hypot(this.x, this.y);
  }
  
  // to make copy of the vector so we can manipulate this copied vector without changing the original one
  copy() {
    return new Vector(this.x, this.y);
  }
  
  // to find a vector with magnitude 1 that points in the same direction
  unit() {
    return this.copy().mult(1 / this.mag());
  }
  
  // that's it for now
  // we will add more useful methods such as dot product, normal, rotate, lerp, ... later when we need them >_<
  
  
}


// lets see an example 
let playerPos = new Vector(10, 50); // x = 10 and y = 50
let playerVel = new Vector(0.5, 0); // vel in x-direction is 0.5 and that in y-direction is 0 

update(dt) {
  playerVel.mult(0.99); // friction
  playerPos.add(playerVel.copy().mult(dt));
}

Optimization Note

Remember it's always better to reuse the old objects rather than creating new objects (if possible) for high performance.

For instance, I could have used

function main() {
  canvas = createCanvas(320, 180);
  ctx = canvas.getContext("2d");
  
  document.body.append(canvas);
  
  
  now = then = frameThen = performance.now();
  frames = 0;
  fps = 0;
  dt = 0;
  
  loop();
}

function drawText(ctx, text, x = 0, y = 0, size = 12, fontFamily = "Sans-serif") {
  ctx.font = `${size}px ${fontFamily}`;
  ctx.fillStyle = "#fff";
  ctx.textAlign = "center";
  ctx.fillText(text, x, y);
}

function loop() {
  requestAnimationFrame(loop);
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  
  // calculate dt and fps
  now = performance.now();
  dt = (now - then) / 1000;
  if(now - frameThen > 1000) {
    fps = frames;
    frames = 0;
    frameThen = now;
  } else {
    frames += 1;
  }
  then = now;
  
  
  
  
  // render fps
  drawText(ctx, `FPS: ${fps}`, 30, 20)
  
  

  
}

Maximum FPS depends on your device's screen refresh rate

1d array is better for performance than 2d array. However, we're going to use 2d array as it is a lot easier to work with and understand. You can always make transition to 1d once you're comfortable with 2d

Optimization Note

We're going to divide our big tilemap into small chunks (which is nothing but a canvas) say 16 tiles × 16 tiles, do calculations to render all the tiles to their corresponding chunks and store all the chunks. These stored chunks are like images, we would then just draw the visible chunks to our main canvas.

You might ask "why the need of chunk based tilemap? Can't we do it the simple way?" Yes, you can do it in a simple way but that approach gives low performance and not suitable for big tilemaps. That's why we're using chunk based tilemap. 
class Tilemap {
  constructor(tileset, tilesetDim, data, tilesize = 16, chunkSize = 16) {
    this.tileset = tileset; // tileset image
    this.tilesetDim = tilesetDim; // dimension of image. we are using 11×5 image. it's stored as an array [11, 5]
    this.data = data; // JSON data of map
    this.tilesize = tilesize; // actual render size of each tile
    this.chunks = new Map(); // container of canvas for each chunk. We will prerender each chunk then store it so that we can render it without recalculating stuff again
    this.chunkSize = chunkSize; // it determines the size of each chunk. for example if chunkSize is 10 and our tilemap is 50×50 tiles then we would have total 5×5 = 25 chunks (each chunk is 10 tiles × 10 tiles (rows × columns))
    this.cx_max = Math.ceil(this.data[0].length / this.chunkSize); // total number of chunk in x-direction
    this.cy_max = Math.ceil(this.data.length / this.chunkSize); // total number of chunk in y-direction
    
  }
  
  // init() is only a one time function. Once we prerender all chunks, we don't have to recalculate everything and can just draw right away.
  init() {
    
    let rows = this.data.length;
    let columns = this.data[0].length;
    
    let sw = this.tileset.width / this.tilesetDim[0]; // source width
    let sh = this.tileset.height / this.tilesetDim[1]; // source height
    
    
    this.data.forEach((row, rIdx) => {
      row.forEach((tile, cIdx) => {
        
        let chunk_y = Math.floor(rIdx / this.chunkSize); // y coordinate of chunk such as 0, 1, 2, 3 (we have only 4 chunks in y-direction remember)
        
        // Note: cx refers to chunk x coordinate (in our particular case it could be 0,1,2,3 as each row of tilemap is 50 tiles wide and each chunk is 16 (default) tiles wide). Same goes for cy.
        // first lemme explain how canvases are stored in this.chunks. this.chunks contains cy—Map() pair, each cy stores all possible values of cx—Map() pair and this cx contains our canvas of that chunk in Map();
        /* Think of it as something like:
        this.chunks = new Map(
          0 (cy) -> new Map(
              0 (cx) -> new Map(chunk canvas of (0,0)),
              1 -> new Map(chunk canvas of (1,0)),
              2 -> new Map(chunk canvas of (2,0)),
              3 -> new Map(chunk canvas of (3,0))
            ),
          1 (cy) -> new Map(
              0 (cx) -> new Map(chunk canvas of (0,1)),
              1 -> new Map(chunk canvas of (1,1)),
              2 -> new Map(chunk canvas of (2,1)),
              3 -> new Map(chunk canvas of (3,1))
            ),
            .... and so on!
        )
        */
        if(!this.chunks.has(chunk_y)) this.chunks.set(chunk_y, new Map());
        
        
        let chunk_x = Math.floor(cIdx / this.chunkSize);
        if(!this.chunks.get(chunk_y).has(chunk_x)) {
          let offcanvas = document.createElement("canvas");
          offcanvas.width = offcanvas.height = this.chunkSize * this.tilesize;
          
          this.chunks.get(chunk_y).set(chunk_x, offcanvas);
        }
        
        let chunk_canvas = this.chunks.get(chunk_y).get(chunk_x);
        let ctx = chunk_canvas.getContext("2d");
        
        
        
        let sx = ((tile - 1) % this.tilesetDim[0]) * sw;
        let sy = Math.floor((tile - 1) / this.tilesetDim[0]) * sh;
        
        let dx = (cIdx % this.chunkSize) * this.tilesize;
        let dy = (rIdx % this.chunkSize) * this.tilesize;
        
        ctx.drawImage(this.tileset, sx, sy, sw, sh, dx, dy, this.tilesize, this.tilesize); // draw tile to our chunk
        
        
      })
    })
    
  }
  
  render(ctx) {
    
    
    // render all chunks in game loop function. I know it's not a good idea to draw all chunks. We'll draw only visible chunks in next tutorial. For now let's just enjoy >_<
    for(let cy = 0; cy < this.cy_max; cy++) {
      for(let cx = 0; cx < this.cx_max; cx++) {
        let chunk = this.chunks.get(cy).get(cx); // chunk canvas
        ctx.drawImage(chunk, 0, 0, chunk.width, chunk.height, cx * chunk.width, cy * chunk.height, chunk.width, chunk.height); // draw chunk to our main canvas
        
      }
    }
    
  }
  
}
